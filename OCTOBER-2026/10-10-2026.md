#include <vector>
#include <algorithm>
#include <cmath>

using namespace std;

class Solution {
public:
    long long minSumSquareDiff(vector<int>& nums1, vector<int>& nums2, int k1, int k2) {
        int n = nums1.size();
        vector<long long> diff(n);
        
        // Step 1: Calculate absolute differences
        for (int i = 0; i < n; i++) {
            diff[i] = abs((long long)nums1[i] - nums2[i]);
        }
        
        // Step 2: Sort in descending order
        sort(diff.begin(), diff.end(), greater<long long>());
        
        // Append 0 as a sentinel for the case where all differences can be reduced to 0
        diff.push_back(0);
        
        long long total_ops = (long long)k1 + k2;
        long long count = 1; // Number of elements currently being reduced to the same level
        
        for (int i = 0; i < n; i++) {
            // Cost to reduce the current 'count' elements down to diff[i+1]
            long long cost = count * (diff[i] - diff[i+1]);
            
            if (total_ops >= cost) {
                // We have enough operations to bridge this gap
                total_ops -= cost;
                count++;
            } else {
                // We don't have enough operations to bridge the gap
                // Distribute the remaining operations among the 'count' elements
                long long q = total_ops / count;
                long long r = total_ops % count;
                
                long long ans = 0;
                
                // (count - r) elements are reduced by q
                long long val1 = diff[i] - q;
                ans += (count - r) * val1 * val1;
                
                // r elements are reduced by q + 1
                long long val2 = diff[i] - q - 1;
                ans += r * val2 * val2;
                
                // Add the squares of the remaining untouched elements
                for (int j = i + 1; j < n; j++) {
                    ans += diff[j] * diff[j];
                }
                
                return ans;
            }
        }
        
        // If the loop completes, all differences have been reduced to 0
        return 0;
    }
};
